# 云端采集定时表
# . env.sh  -> 载入容器环境变量（cron 默认不继承 docker -e）
# flock -n   -> 上一轮没跑完则跳过，避免重叠
SHELL=/bin/bash

# 1) 小红书采集：每 20 分钟（:00/:20/:40）
*/20 * * * * cd /app/cloud && . /app/cloud/env.sh && flock -n /tmp/xhs.lock /usr/local/bin/python run_batch.py >> /app/data/cron.log 2>&1

# 2) 电话补齐：每小时 :05/:25/:45（与 XHS 采集错峰 5 分钟）
5,25,45 * * * * cd /app/cloud && . /app/cloud/env.sh && flock -n /tmp/phone.lock /usr/local/bin/python cloud_phone_fill.py >> /app/data/phone_fill.log 2>&1

# 3) 坐标补齐：每天凌晨 3:00（量小，全量重试）
0 3 * * * cd /app/cloud && . /app/cloud/env.sh && flock -n /tmp/coord.lock /usr/local/bin/python cloud_coord_fill.py >> /app/data/coord_fill.log 2>&1

# 4) 营业时间补齐：每天凌晨 4:00
0 4 * * * cd /app/cloud && . /app/cloud/env.sh && flock -n /tmp/hours.lock /usr/local/bin/python cloud_hours_fill.py >> /app/data/hours_fill.log 2>&1

# 5) 高德全字段采集（评分/人均/电话/营业时间/坐标，一次 place/text）：每小时 :15/:35/:55（错峰）
15,35,55 * * * * cd /app/cloud && . /app/cloud/env.sh && flock -n /tmp/amap_fill.lock /usr/local/bin/python cloud_amap_fill.py --apply >> /app/data/amap_fill.log 2>&1
